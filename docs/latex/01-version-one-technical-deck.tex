% 0. SYSTEM ARCHITECTURE
%======================================================================
\gateheader{System Architecture}{End-to-end view}{Browser \textbullet{} Next.js 16 on Vercel \textbullet{} Vercel Workflows \textbullet{} AI Gateway \textbullet{} Supabase + RLS \textbullet{} Sentry}{Master diagram}

\figureslot{00-system-architecture.png}{
\begin{techbox}[Stack at a glance]
\setlist[itemize]{leftmargin=12pt,itemsep=2pt}
\begin{itemize}
\item \kw{App} Next.js 16 App Router on Vercel Fluid Compute
\item \kw{Auth} Supabase Auth (\code{@supabase/ssr}) cookies-based SSR
\item \kw{DB} Supabase Postgres + RLS via Drizzle \code{pgPolicy}
\item \kw{Orchestration} Vercel Workflows (GA April 2026)
\item \kw{AI} AI SDK v6 \textbullet{} \code{Output.object()} \textbullet{} \code{ToolLoopAgent}
\item \kw{Models} GPT-5.4 \textbullet{} Sonnet 4.6 \textbullet{} Haiku 4.5 \textbullet{} Gemini 2.5 Flash
\item \kw{Images} \code{gpt-image-1} + Flux Ultra 1.1 (Replicate)
\item \kw{Observability} Sentry + \code{@sentry/opentelemetry}
\item \kw{Verify} Playwright + Chrome DevTools MCP
\end{itemize}
\end{techbox}

\begin{techbox}[Why this shape]
\setlist[itemize]{leftmargin=12pt,itemsep=2pt}
\begin{itemize}
\item One Next.js app today, monorepo packages already split for future apps
\item RLS at the DB \textit{enforces} multi-tenancy --- no client trust
\item Workflows make the multi-step AI pipeline durable and idempotent
\item Mock provider on every AI task = zero-network demo safety
\end{itemize}
\end{techbox}
}

% 3. GATE B
%======================================================================
\gateheader{Gate B \textbullet{} Tenancy \textbullet{} Credits \textbullet{} Admin}{4--5 days}{RLS at the DB. Immutable ledger. Lazy Super Admin queue.}{v0.2-gate-b-tenancy-credits-pass}

\figureslot{03-gate-b-tenancy-credits-admin.png}{
\begin{techbox}[Three pillars]
\setlist[itemize]{leftmargin=12pt,itemsep=2pt}
\begin{itemize}
\item \kw{Tenancy schema}: \code{profiles}, \code{organizations}, \code{organization\_members}, \code{admin\_actions}, \code{memberships} --- every table guarded by Drizzle \code{pgPolicy}
\item \kw{Credits ledger}: \code{credit\_balances} (materialized) + \code{credit\_ledger} (append-only); helpers \code{reserveCredits} / \code{deductCredits} / \code{refundCredits} / \code{grantCredits}
\item \kw{Lazy Super Admin}: pending-user queue, credit-grant form, audit log --- minimal pages, server-side guard checked on every privileged action
\end{itemize}
\end{techbox}

\begin{techbox}[Invariants]
\setlist[itemize]{leftmargin=12pt,itemsep=1pt}
\begin{itemize}
\item $\sum \text{ledger.amount per org} = \text{credit\_balances.balance}$
\item No row reachable across orgs (cross-tenant probes fail)
\end{itemize}
\end{techbox}

\begin{donewhen}
Pending user blocked \textbullet{} approved user lands in builder \textbullet{} ledger sum equals balance for every org \textbullet{} cross-tenant tests pass.
\end{donewhen}
}

% 5. GATE D
%======================================================================
\gateheader{Gate D \textbullet{} Store Generation Pipeline}{6--7 days}{Vercel Workflow \code{generation.run} \textbullet{} 9 retryable steps \textbullet{} credits deducted only on success}{v0.4-gate-d-generation-pass}

\figureslot{05-gate-d-generation-workflow.png}{
\begin{techbox}[The nine steps]
\setlist[enumerate]{leftmargin=14pt,itemsep=1pt}
\begin{enumerate}\small
\item Load Context (brief + tenant + ContextEnvelope)
\item Reserve Credits ($\to$ ledger row)
\item \kw{createStorePlan} (GPT-5.4 \textbullet{} \code{Output.object()})
\item \kw{generateCatalog} (12 products)
\item \kw{generateCopyBlocks} (Sonnet 4.6)
\item \kw{generateProductImages} (parallel \textbullet{} mock fallback)
\item Persist Artifacts ($\to$ \code{store\_artifacts} + Storage)
\item \kw{summarizeVerification} (after Gate E pass)
\item Deduct or Refund (terminal status)
\end{enumerate}
\end{techbox}

\begin{techbox}[Artifact bus]
\code{store.config.json} \textbullet{} \code{theme.tokens.json} \textbullet{} \code{catalog.seed.json} \textbullet{} \code{copy.blocks.json} \textbullet{} \code{generation.report.json}\\[2pt]
Template registry: 1 strong base + 6 presets (fashion, beauty, electronics, home, fitness, gifts).
\end{techbox}

\begin{donewhen}
Brief reaches \code{generation.artifacts\_created} reliably \textbullet{} credits deducted only on success \textbullet{} mock-provider end-to-end in $<$10 s \textbullet{} every artifact validates against its Zod schema.
\end{donewhen}
}

\figureslot{06-gate-e-storefront-verification.png}{
\begin{techbox}[Render path]

\begin{techbox}[Verification runner --- 9 checks]
\setlist[enumerate]{leftmargin=14pt,itemsep=1pt}

\item Tenant ownership probe (impersonate other org)
\item Credit-ledger invariant
\item All required routes return 200
\item Cart add/remove/qty/subtotal (Playwright)
\item Mock checkout success + cancel
\item Responsive @ 360 / 768 / 1280 / 1920
\item Playwright a11y axe scan
\item Prompt-injection / secret-leak grep on artifacts
\end{enumerate}
\end{techbox}

\begin{donewhen}
100\% of verification checks pass \textbullet{} failed verification keeps store out of \code{demo\_ready} (credits refunded).
\end{donewhen}

\gateheader{Bonus \textbullet{} AI Task Routing Matrix}{Per-task primary model + documented fallback + always-on mock}{Refs: docs/decisions/0004-ai-task-routing-matrix.md}{Reference}

\figureslot{08-bonus-ai-routing-matrix.png}{
\begin{techbox}[Routing rules summary]
\setlist[itemize]{leftmargin=12pt,itemsep=1pt}
\begin{itemize}\small
\item Reasoning-heavy structured tasks $\to$ \kw{Claude Sonnet 4.6}

\item Cheap utility classification $\to$ \kw{Claude Haiku 4.5}
\item Photoreal images $\to$ \kw{Flux Ultra 1.1} on Replicate
\item Composition/text-on-image $\to$ \kw{gpt-image-1}
\item Catalog scale-out fallback $\to$ \kw{Gemini 2.5 Flash} (cheapest)
\item \textbf{Mock provider} drives offline demo for every task
\end{itemize}
\end{techbox}

\begin{techbox}[Why two image models]
\code{gpt-image-1} composes layouts and renders text on labels reliably; Flux Ultra 1.1 wins on photoreal hero shots. Routing picks based on the artifact \code{image.kind}: \code{hero}, \code{lifestyle}, \code{packshot}, \code{label}.
\end{techbox}
